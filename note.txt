                $barcodePath = IMGBS . 'barcode/member_' . $memberID . '.png';

                $html = '<div>';
                $html .= '<strong>' . $memberID . '</strong>';

                if (file_exists($barcodePath)) {

                }

                $html .= '</div>';

                return $html;
